#!/bin/bash
set -e

PERSISTENT=/app/persistent

# Directorios a persistir: "ruta_real:ruta_en_volumen"
DIRS=(
  "/app/uploads:uploads"
  "/app/whatsapp-sessions:whatsapp-sessions"
  "/app/data:data"
  "/app/volumes:volumes"
)

echo "[entrypoint] Configurando directorios persistentes..."

for entry in "${DIRS[@]}"; do
  APP_PATH="${entry%%:*}"
  VOL_PATH="$PERSISTENT/${entry##*:}"

  # Crear carpeta en el volumen si no existe
  mkdir -p "$VOL_PATH"

  # Si la ruta original existe y NO es symlink, borrarla
  if [ -e "$APP_PATH" ] && [ ! -L "$APP_PATH" ]; then
    rm -rf "$APP_PATH"
  fi

  # Crear symlink apuntando al volumen
  ln -sfn "$VOL_PATH" "$APP_PATH"

  echo "[entrypoint] $APP_PATH -> $VOL_PATH"
done

echo "[entrypoint] Listo. Iniciando app..."
exec node dist/index.js
